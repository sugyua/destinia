cmake_minimum_required(VERSION 3.16)
project(destinia_host CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
pkg_check_modules(SDL2_MIXER SDL2_mixer)
pkg_check_modules(UNICORN REQUIRED unicorn)
add_executable(destinia_host main.cpp)
target_include_directories(destinia_host PRIVATE ${SDL2_INCLUDE_DIRS} ${UNICORN_INCLUDE_DIRS})
target_link_directories(destinia_host PRIVATE ${SDL2_LIBRARY_DIRS} ${UNICORN_LIBRARY_DIRS})
target_link_libraries(destinia_host PRIVATE ${SDL2_LIBRARIES} ${UNICORN_LIBRARIES})
if(SDL2_MIXER_FOUND)
  target_compile_definitions(destinia_host PRIVATE HAS_SDL_MIXER=1)
  target_include_directories(destinia_host PRIVATE ${SDL2_MIXER_INCLUDE_DIRS})
  target_link_directories(destinia_host PRIVATE ${SDL2_MIXER_LIBRARY_DIRS})
  target_link_libraries(destinia_host PRIVATE ${SDL2_MIXER_LIBRARIES})
endif()
